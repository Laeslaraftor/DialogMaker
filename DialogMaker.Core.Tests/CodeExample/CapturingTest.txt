void LoopFunction()
{
	 int minValue;
                int maxValue;
                string textNumber;

                do
                {
                    Console.Write("Enter min value: ");
                    textNumber = Console.ReadLine();
                } 
                while (!int.TryParse(textNumber, out minValue));
                do
                {
                    Console.Write("Enter max value: ");
                    textNumber = Console.ReadLine();
                } 
                while (!int.TryParse(textNumber, out maxValue));

                IEnumerable<int> GetValuesWhileNotInRange()
                {
                    var random = Random.Shared;

                    int GetRandomNumber()
                    {
                        return random.Next(0, 100);
                    }

                    while (true)
                    {
                        var value = GetRandomNumber();

                        if (value >= minValue && value <= maxValue)
                        {
                            yield break;
                        }

                        yield return value;
                    }
                }

                Console.WriteLine("Printing numbers 0-100 while it not in range " + minValue + "-" + maxValue + ":");

                foreach (var number in GetValuesWhileNotInRange())
                {
                    Console.WriteLine(number);
                }
}
